    70→    70→// Utility functions
    71→    71→function formatDuration(seconds: number): string {
    72→    72→  if (!seconds || isNaN(seconds)) return "0:00";
    73→    73→  const h = Math.floor(seconds / 3600);
    74→    74→  const m = Math.floor((seconds % 3600) / 60);
    75→    75→  const s = seconds % 60;
    76→    76→  if (h > 0) return `${h}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    77→    77→  return `${m}:${s.toString().padStart(2, "0")}`;
    78→    78→}
    79→    79→
    80→    80→function formatFileSize(bytes: number): string {
    81→    81→  if (!bytes) return "0 B";
    82→    82→  const sizes = ["B", "KB", "MB", "GB"];
    83→    83→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
    84→    84→  return `${(bytes / Math.pow(1024, i)).toFixed(1)} ${sizes[i]}`;
    85→    85→}
    86→    86→
    87→    87→function formatDate(dateStr: string): string {
    88→    88→  const date = new Date(dateStr);

    90→    90→    month: "short",
    91→    91→    day: "numeric",
    92→    92→    hour: "2-digit",
    93→    93→    minute: "2-digit",
    94→    94→  });
    95→    95→}
