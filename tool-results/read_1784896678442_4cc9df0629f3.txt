    70→// Utility functions
    71→function formatDuration(seconds: number): string {
    72→  if (!seconds || isNaN(seconds)) return "0:00";
    73→  const h = Math.floor(seconds / 3600);
    74→  const m = Math.floor((seconds % 3600) / 60);
    75→  const s = seconds % 60;
    76→  if (h > 0) return `${h}:${m.toString().padStart(2, "0")}:${s.toString().padStart(2, "0")}`;
    77→  return `${m}:${s.toString().padStart(2, "0")}`;
    78→}
    79→
    80→function formatFileSize(bytes: number): string {
    81→  if (!bytes) return "0 B";
    82→  const sizes = ["B", "KB", "MB", "GB"];
    83→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
    84→  return `${(bytes / Math.pow(1024, i)).toFixed(1)} ${sizes[i]}`;
    85→}

    87→function formatDate(dateStr: string): string {
    88→  const date = new Date(dateStr);
    89→  return date.toLocaleDateString("en-US", {
    90→    month: "short",
    91→    day: "numeric",
    92→    hour: "2-digit",
    93→    minute: "2-digit",
    94→  });
    95→}

   454→  const mp4Qualities = ["360", "480", "720", "1080"];

   457→  // Poll for download status updates

   460→      try {
   461→        const res = await fetch(`/api/download/${id}`);

   466→          prev.map((d) =>
   467→            d.id === id
   468→              ? {
   469→                  ...d,
   470→                  status: data.status,
   471→                  progress: data.progress,
   472→                  fileSize: data.fileSize || d.fileSize,
   473→                  error: data.error,
   474→                }
   475→              : d
   476→          )
   477→        );
   478→
   479→        // Stop polling when done
   480→        if (data.status === "completed") {
   481→          clearInterval(interval);
   482→          pollingRef.current.delete(id);
   483→          toast.success("Download ready!", {
   484→            description: "Your file is ready to download.",
   485→          });
   486→          fetchHistory();
   487→        } else if (data.status === "failed") {
   488→          clearInterval(interval);
   489→          pollingRef.current.delete(id);
   490→          if (data.error && data.error !== "Cancelled") {
   491→            toast.error("Download failed", {
   492→              description: data.error,
   493→            });
   494→          }
   495→        }
   496→      } catch {
   497→        // silent fail
   498→      }
   499→    }, 2000);
   500→    pollingRef.current.set(id, interval);
   501→  }, []);
   502→
   503→  // Cleanup polling on unmount
   504→  useEffect(() => {
   505→    return () => {
   506→      pollingRef.current.forEach((interval) => clearInterval(interval));
   507→    };
   508→  }, []);
   509→
   510→  // Fetch history on mount
   511→  useEffect(() => {
   512→    fetchHistory();
   513→  }, []);
   514→
   515→  const fetchHistory = async () => {
   516→    try {
   517→      const res = await fetch("/api/history");
   518→      if (res.ok) {
   519→        const data = await res.json();
   520→        setHistory(data);
   521→      }
   522→    } catch {
   523→      // silent fail
   524→    }
   525→  };
   526→
   527→  const extractVideoId = (inputUrl: string): string | null => {
   528→    const match = inputUrl.match(
   529→      /(?:youtube\.com\/(?:watch\?v=|shorts\/|embed\/|v\/|live\/)|youtu\.be\/)([a-zA-Z0-9_-]{11})/
   530→    );
   531→    return match ? match[1] : null;
   532→  };
   533→
   534→  const handleFetchVideoInfo = useCallback(async () => {
   535→    const videoId = extractVideoId(url);
   536→    if (!videoId) {
   537→      toast.error("Invalid URL", {
   538→        description: "Please enter a valid YouTube URL.",
   539→      });
   540→      return;
   541→    }
   542→
   543→    setIsLoadingVideo(true);
   544→    setVideoInfo(null);
   545→    setHasStarted(true);
   546→
   547→    try {
   548→      const res = await fetch("/api/video-info", {
   549→        method: "POST",
   550→        headers: { "Content-Type": "application/json" },
   551→        body: JSON.stringify({ url }),
   552→      });
   553→
   554→      if (!res.ok) {
   555→        const errorData = await res.json();
   556→        const code = errorData.code;
   557→        if (code === "BOT_BLOCKED") {
   558→          toast.error("YouTube is blocking requests", {
   559→            description: "This video might be restricted. Try a different video or wait a moment and retry.",
   560→            duration: 8000,

   562→              label: "Retry",
   563→              onClick: () => handleFetchVideoInfo(),
   564→            },
   565→          });
   566→          return;
   567→        }
   568→        throw new Error(errorData.error || "Failed to fetch video info");
   569→      }
   570→
   571→      const data = await res.json();
   572→      setVideoInfo(data);
   573→    } catch (error: any) {
   574→      toast.error("Failed to analyze video", {
   575→        description: error.message || "Something went wrong. Please try again.",
   576→        duration: 6000,
   577→      });
   578→    } finally {
   579→      setIsLoadingVideo(false);
   580→    }

   583→  const handleStartDownload = useCallback(async () => {

   592→        headers: { "Content-Type": "application/json" },
   593→        body: JSON.stringify({
   594→          url,
   595→          format,
   596→          quality,
   597→          title: videoInfo.title,
   598→          thumbnail: videoInfo.thumbnail,
   599→          duration: videoInfo.duration,
   600→          channel: videoInfo.channel,
   601→        }),

   606→      }

   608→      const taskData = await res.json();

   610→      // Add to active downloads

   612→        id: taskData.id,
   613→        url,
   614→        title: videoInfo.title,
   615→        thumbnail: videoInfo.thumbnail,
   616→        duration: videoInfo.duration,
   617→        format,
   618→        quality,
   619→        status: "processing",
   620→        progress: 0,
   621→        fileSize: null,
   622→        createdAt: new Date().toISOString(),
   623→      };
   624→
   625→      setActiveDownloads((prev) => [newDownload, ...prev]);
   626→
   627→      // Start download via worker proxy API
   628→      fetch("/api/worker/download", {
   629→        method: "POST",
   630→        headers: { "Content-Type": "application/json" },
   631→        body: JSON.stringify({
   632→          id: taskData.id,
   633→          url,
   634→          format,
   635→          quality,
   636→        }),
   637→      }).catch(() => {
   638→        toast.error("Error", { description: "Failed to start download worker." });
   639→      });
   640→
   641→      // Start polling for progress
   642→      startPolling(taskData.id);
   643→
   644→      toast.info("Download started", {
   645→        description: `Converting to ${format.toUpperCase()} ${format === "mp4" ? `${quality}p` : `${quality}kbps`}...`,
   646→      });
   647→
   648→      // Reset form
   649→      setUrl("");
   650→      setVideoInfo(null);
   651→      inputRef.current?.focus();
   652→    } catch (error: any) {
   653→      toast.error("Error", {
   654→        description: error.message || "Failed to start download.",
   655→      });
   656→    } finally {
   657→      setIsConverting(false);
   658→    }
   659→  }, [videoInfo, url, format, quality]);

   677→      // Clipboard access denied

   685→      }
   686→    },
   687→    [handleFetchVideoInfo, url, videoInfo]
   688→  );

   703→            <span className="text-lg font-bold text-foreground tracking-tight">VortexTube</span>

   719→                  </Badge>

   721→                <ChevronDown className={cn("w-3 h-3 transition-transform", showHistory && "rotate-180")} />
   722→              </Button>
   723→            )}
   724→          </div>
   725→        </div>
   726→      </header>
   727→
   728→      {/* Main Content */}
   729→      <main className="relative z-10 flex-1 px-6 pb-20">

   749→              YouTube to{" "}

   795→                      onClick={handleFetchVideoInfo}
   796→                      disabled={!url || isLoadingVideo}
   797→                    >
   798→                      {isLoadingVideo ? (
   799→                        <Loader2 className="w-4 h-4 animate-spin" />
   800→                      ) : hasStarted ? (
   801→                        "Fetch"
   802→                      ) : (

   804→                          Analyze
   805→                          <ArrowRight className="w-4 h-4" />

   808→                    </Button>

   811→
   812→                {/* Video Preview */}
   813→                <AnimatePresence>
   814→                  {(videoInfo || isLoadingVideo) && (

   823→                      </div>
   824→
   825→                      {/* Format Selection + Download Button */}
   826→                      {!isLoadingVideo && videoInfo && (
   827→                        <motion.div

   834→                            setFormat={setFormat}

   836→                            setQuality={setQuality}
   837→                            mp4Qualities={mp4Qualities}
   838→                            mp3Qualities={mp3Qualities}

   840→
   841→                          <Button
   842→                            size="lg"
   843→                            className="gap-2.5 bg-gradient-to-r from-emerald-600 to-teal-600 hover:from-emerald-700 hover:to-teal-700 text-white font-semibold shadow-xl shadow-emerald-500/25 transition-all hover:shadow-emerald-500/40 hover:scale-[1.02] active:scale-[0.98]"
   844→                            onClick={handleStartDownload}

   861→                    </motion.div>

   896→                          .filter((d) => d.status === "completed")
   897→                          .map((item) => (
   898→                            <DownloadProgressItem key={item.id} item={item} onCancel={handleCancelDownload} />
   899→                          ))}
   900→                      </AnimatePresence>
   901→                    </div>

   904→              </motion.div>

   907→
   908→          {/* Features Section */}
   909→          {!hasStarted && (
   910→            <motion.div
   911→              initial={{ opacity: 0, y: 30 }}
   912→              animate={{ opacity: 1, y: 0 }}
   913→              transition={{ duration: 0.6, delay: 0.5 }}
   914→              className="mt-14"
   915→            >
   916→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   917→                <FeatureCard
   918→                  icon={Zap}
   919→                  title="Lightning Fast"
   920→                  description="Convert videos in seconds with our optimized processing engine. No waiting, instant results."
   921→                />
   922→                <FeatureCard
   923→                  icon={Shield}
   924→                  title="Safe & Private"
   925→                  description="No registration required. Your data stays private. No tracking, no ads, no malware."
   926→                />
   927→                <FeatureCard
   928→                  icon={Music}

   930→                  description="Download in up to 1080p for video and 320kbps for audio. Choose the quality that suits you."
   931→                />

   949→                        <History className="w-4 h-4" />
   950→                        Download History
   951→                      </h2>
